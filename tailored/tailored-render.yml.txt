name: Tailored 렌더
# tailored/request.txt 에 "<에피소드 폴더> [텔레그램 설명]" 한 줄을 적어 푸시하면
# tailored_render.py 로 만화 컷을 렌더(효과음·편별 로파이는 ElevenLabs로 1회 생성해 캐시) → mp4·소리 커밋 → 텔레그램 전송. 발행은 하지 않는다.
on:
  push:
    paths: ["tailored/request.txt"]
  workflow_dispatch:
    inputs:
      args:
        description: "<에피소드 폴더>"
        required: true
        default: "tailored/episodes/s1e01"

permissions:
  contents: write

jobs:
  render:
    runs-on: ubuntu-latest
    env:
      TZ: Asia/Seoul
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.12" }
      - name: 의존성
        run: |
          sudo apt-get update -qq && sudo apt-get install -y -qq ffmpeg fonts-noto-cjk
          pip install -q requests pillow
      - name: 렌더 + 텔레그램
        env:
          ELEVEN_API_KEY: ${{ secrets.ELEVEN_API_KEY }}
          TELEGRAM_TOKEN: ${{ secrets.TELEGRAM_TOKEN }}
          TELEGRAM_CHAT_ID: ${{ secrets.TELEGRAM_CHAT_ID }}
        run: |
          LINE="${{ inputs.args }}"
          if [ "${{ github.event_name }}" = "push" ]; then LINE="$(head -1 tailored/request.txt)"; fi
          EP=$(echo "$LINE" | awk '{print $1}'); NOTE=$(echo "$LINE" | cut -d' ' -f2-)
          python tailored_render.py "$EP"
          OUT=$(ls "$EP"/*.mp4 | grep -v '/c[0-9]' | head -1)
          python -c "import sys; from common import telegram_video; telegram_video(sys.argv[1], sys.argv[2])" "$OUT" "[Tailored · Actions 렌더] $NOTE"
      - name: 결과 커밋
        if: ${{ success() }}
        run: |
          git config user.name casto-bot && git config user.email casto@solvup.local
          for p in tailored/episodes/*/*.mp4 tailored/sfx assets/bgm; do [ -e "$p" ] && git add -f "$p"; done; true
          if git diff --cached --quiet; then echo "변경 없음"; else
            git commit -m "tailored render: $(head -1 tailored/request.txt 2>/dev/null || echo manual) [skip ci]" && git pull --rebase --autostash origin main && git push origin main
          fi
